function delTask(seq) {
  const t = tasks.find(x => x.seq === seq) || tasks.find(x => String(x.seq) === String(seq));
  if (!t) return alert("未找到该记录，请刷新页面后重试");
  if (!confirm(`确定删除任务 #${t.seq}？\n${(t.desc || "").slice(0, 40)}…`)) return;
  /* 用字符串比较，避免 seq 是文本型时删不掉 */
  tasks = tasks.filter(x => String(x.seq) !== String(t.seq));
  saveLocal(); render();
}

  空值保护: true
